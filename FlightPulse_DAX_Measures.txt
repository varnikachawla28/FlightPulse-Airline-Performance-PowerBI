FlightPulse_DAX_Measures - PowerBI

DAX Measures 
1. Airline Revenue Contribution % = DIVIDE([Total Revenue],[Total Revenue All Airlines],0)
2. Airline Revenue Rank = RANKX ( ALL ( Dim_Airline[Airline_Name] ), [Total Revenue],DESC, DENSE)
3. Average Delay = AVERAGE(Fact_Flights[Delay_Minutes])
4. Average Flight Distance = AVERAGE(Fact_Flights[Distance_KM] )
5. Average Passenger Per Flight = DIVIDE([Total Passengers],[Total Flights],0)
6. Average Ticket Price = AVERAGE(Fact_Flights[Average_Ticket_Price])
7. Cost Per Flight = DIVIDE([Total Operating Cost],[Total Flights],0)
8. Delay Rate % = DIVIDE([Delayed Flight],[Total Flights],0)
9. Delayed Flight = CALCULATE([Total Flights],Fact_Flights[Delay_Category] <> "On Time")
10. Flight Efficiency Score = [On Time Performance %]-[Delay Rate %]
11. Flights Delayed Over 2 Hours = CALCULATE([Total Flights],Fact_Flights[Delay_Minutes] > 120) 
12. Month Over Month Growth % = DIVIDE([Total Revenue] - [Previous Month Revenue],[Previous Month Revenue],0)
13. Month To Date Revenue = TOTALMTD([Total Revenue],Dim_Date[Date])
14. On Time Flight = CALCULATE([Total Flights],Fact_Flights[Delay_Category]="On Time")
15. On Time Performance % = DIVIDE([On Time Flight],[Total Flights],0)
16. Previous Month Revenue = CALCULATE([Total Revenue],DATEADD(Dim_Date[Date],-1,MONTH ))
17. Previous Year Profit = CALCULATE([Total Profit],SAMEPERIODLASTYEAR(Dim_Date[Date]))
18. Previous Year Revenue = CALCULATE([Total Revenue],SAMEPERIODLASTYEAR(Dim_Date[Date]))
19. Profit Margin % = DIVIDE([Total Profit],[Total Revenue],0)
20. Profit Per Flight = DIVIDE([Total Revenue],[Total Flights],0)
21. Profit Year Over Year Growth % = DIVIDE([Total Profit]-[Previous Year Profit],[Previous Year Profit],0)
22. Quarter To Date Revenue = TOTALQTD([Total Revenue],Dim_Date[Date])
23. Revenue Per Passenger = DIVIDE([Total Revenue],[Total Passengers],0)
24. Revenue Year Over Year Change = [Total Revenue] - [Previous Year Revenue]
25. Revenue Year Over Year Growth % = DIVIDE([Revenue Year Over Year Change],[Previous Year Revenue],0)
26. Severe Delay Rate % = DIVIDE([Severe Delays],[Total Flights],0)
27. Severe Delays = CALCULATE([Total Flights],Fact_Flights[Delay_Category]= "Severe Delays")
28. Top 5 Airline Flag = IF([Airline Revenue Rank]<=5 , 1 ,0 )
29. Total Flights = DISTINCTCOUNT(Fact_Flights[Flight_ID])
30. Total Operating Cost = SUM(Fact_Flights[Operating_Cost])
31. Total Passengers = SUM(Fact_Flights[Passenger_Count])
32. Total Profit = SUM(Fact_Flights[Profit])
33. Total Revenue = SUM((Fact_Flights[Ticket_Revenue]))
34. Total Revenue All Airlines = CALCULATE([Total Revenue], REMOVEFILTERS(Dim_Airline[Airline_Name]))
35. Year To Date Profit = TOTALYTD([Total Profit],Dim_Date[Date])
36. Year To Date Revenue = TOTALYTD([Total Revenue],Dim_Date[Date])
